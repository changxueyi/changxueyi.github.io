func main() {
   //声明切片类型
   a := []int{1, 2, 3, 4, 5}
   //s := a[0:3:5]
   /*fmt.Println("s=", s) //3
   fmt.Println("len(s)=", len(s)) //3-0
   fmt.Println("cap(s)=", cap(s)) //容量为 5-0*/

   s := a[1:4:5]
   fmt.Println("s=",s) //3
   fmt.Println("len(s)=", len(s)) //4-1  长度是：4-1
   fmt.Println("cap(s)=", cap(s)) //5-1  容量为最后一个5-第一个1 = 4
}
