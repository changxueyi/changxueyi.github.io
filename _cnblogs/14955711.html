package main

import (
    "fmt"
)

type Student struct {
    id    int
    name  string
    score float64
}

func main() {
    s := []Student{
        Student{
            1,
            "yy",
            82,
        },
        Student{
            2,
            "yang",
            67,
        },
        Student{
            3,
            "go",
            91,
        },
    }
    fmt.Println(s)
}
