没什么好说的， 见一个，熟悉一个：

package main
 
import (
    "fmt"
)
 
func main() {
    func() {
        fmt.Printf("func 1\n")
    }() //输出:func 1
 
    func(x int) {
        fmt.Printf("func 2, x is %d\n", x)
    }(2) //输出:func 2, x is 2
 
    a := func(x int) int {
            fmt.Printf("func 3, x is %d\n", x)
            return 5
        } //输出func 3, x is 3
　　　　　　　输出5
    fmt.Println(a(3))
 
    fmt.Printf("%T\n",     func() {
        fmt.Printf("func 1\n")
    }) //func()
 
    fmt.Printf("%T\n",     func(x int) {
        fmt.Printf("func 2, x is %d\n", x)
    }) //输出func(int)
 
    fmt.Printf("%T\n", a) //输出 func(int) int
}
